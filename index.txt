        // Tüm Katalog Görünümü
        function katalogGoster(liste) {
            document.getElementById('katalogAlani').classList.remove('hidden');
            const kutu = document.getElementById('urunListesi');
            kutu.innerHTML = liste.map(u => `
                <a href="?urun=${u.kod}" class="group bg-white rounded-2xl border border-slate-200/80 overflow-hidden shadow-sm hover:shadow-md transition flex flex-col">
                    <div class="aspect-[4/3] bg-slate-100 overflow-hidden relative">

                    </div>
                    <div class="p-4 flex-grow flex flex-col justify-between">
                        <div>
                            <div class="text-[11px] text-slate-400 font-semibold uppercase">${u.kategori}</div>
                            <h3 class="font-bold text-slate-900 text-base mt-0.5 mb-2">${u.ad}</h3>
                        </div>

                            📐 ${u.olcu}
                        </div>
                    </div>
                </a>
            `).join('');

        function kategorileriDoldur() {
            const katSecim = document.getElementById('kategoriSecim');
            const kategoriler = [...new Set(URUNLER.map(u => u.kategori))];
            kategoriler.forEach(k => {
                katSecim.innerHTML += `<option value="${k}">${k}</option>`;
            });
        }

        function urunleriFiltrele() {
            const arama = document.getElementById('aramaKutusu').value.toLowerCase();
            const kat = document.getElementById('kategoriSecim').value;
            const filtreli = URUNLER.filter(u => {
                const aramaUyumu = u.ad.toLowerCase().includes(arama) || u.kod.toLowerCase().includes(arama);
                const katUyumu = kat === "Tümü" || u.kategori === kat;
                return aramaUyumu && katUyumu;
            });
            katalogGoster(filtreli);
        }

        // QR Etiket Üretici Fonksiyonları
        function qrModalAc() {
            document.getElementById('qrModal').classList.remove('hidden');
            const secim = document.getElementById('qrUrunSecim');
            secim.innerHTML = URUNLER.map(u => `<option value="${u.kod}">${u.kod} - ${u.ad}</option>`).join('');
            qrUret();
        }
        function qrModalKapat() { document.getElementById('qrModal').classList.add('hidden'); }
        function qrUret() {
            const kod = document.getElementById('qrUrunSecim').value;
            const urun = URUNLER.find(u => u.kod === kod);
            const qrKutu = document.getElementById('qrcode');
            qrKutu.innerHTML = "";
            const tamLink = window.location.origin + window.location.pathname + "?urun=" + kod;
            new QRCode(qrKutu, { text: tamLink, width: 140, height: 140 });
            document.getElementById('qrEtiketKod').innerText = urun.kod;
            document.getElementById('qrEtiketAd').innerText = `${urun.ad} (${urun.olcu})`;
        }
